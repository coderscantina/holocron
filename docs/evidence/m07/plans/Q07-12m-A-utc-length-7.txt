CREATE OR REPLACE TEMP TABLE jy_steps AS
             WITH v AS (
                SELECT m.session_id, coalesce(e.path, '') AS v, m.event_time AS t, m.seq
                FROM session_members m
                JOIN events e ON e.seq = m.seq AND e.site = ? AND e.kind = 1 AND e.event_time >= epoch_ms(?) AND e.event_time <= epoch_ms(?) WHERE m.site = ? AND m.role = 1 AND m.event_time >= epoch_ms(?) AND m.event_time <= epoch_ms(?) AND m.session_id IN (SELECT session_id FROM jy_entry)
             ), compressed AS (
                SELECT session_id, v, t, seq FROM (
                    SELECT *, lag(v) OVER (PARTITION BY session_id ORDER BY t, seq) AS previous FROM v
                ) WHERE previous IS DISTINCT FROM v
             ), numbered AS (
                SELECT session_id, v, row_number() OVER (PARTITION BY session_id ORDER BY t, seq) AS n
                FROM compressed
             )
             SELECT session_id, list(v ORDER BY n) AS steps FROM numbered WHERE n <= 7
             GROUP BY session_id
