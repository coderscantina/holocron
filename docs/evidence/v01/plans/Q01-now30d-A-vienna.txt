-- Q01-now30d-A-vienna, unfiltered, compared with typical_weekday: `holocron-gate explore --plans
-- --compare typical_weekday` on the 10% load, with a scratch copy of Q01-30d-A-vienna over
-- 2026-09-02 to 10-01, so the range contains the clock (2026-10-01 02:00 local). 21 planned
-- statements: the range's event remainder, the baseline window's event remainder (zb, 2026-07-08
-- to 09-24 from rollups), today's eight baseline days up to 02:00 from events (zu0_ to zu7_),
-- today's part of the range (zt_) and two row reads per panel, the range's and one grouped read
-- over every baseline window. Kept: zb, zu0_, and the grouped read of the first panel. The slot
-- column `g` comes from the view zs_ev over zb_ev and zu_ev. Totals, series buckets and span
-- medians are one grouped read each and are not recorded by --plans.

AND role = 1 AND event_time >= epoch_ms(?) AND event_time <= epoch_ms(?)) m ON m.seq = e.seq LEFT JOIN (SELECT * FROM sessions WHERE site = ? AND start_time <= epoch_ms(?) AND last_time >= epoch_ms(?)) s ON s.session_id = m.session_id WHERE e.site = ? AND e.kind IN (1, 2) AND e.event_time >= epoch_ms(?) AND e.event_time <= epoch_ms(?)) SELECT seq, kind, t, daily_id, profile_id, f0, f1, f2, f3, f4 FROM (SELECT r.seq, r.kind, epoch_ms(r.event_time) AS t, r.daily_id, r.profile_id, r.session_id, r.path AS f0, r.referrer_domain AS f1, r.browser AS f2, r.os AS f3, r.device AS f4 FROM r)
